# Debugger-friendly flags for C-Dogs app + game library when GDB is enabled.
# Does not alter ordinary Release/Debug builds with CDOGS_VITA_GDB=OFF.
if(VITA AND CDOGS_VITA_GDB)
	target_compile_definitions(cdogs-sdl PRIVATE CDOGS_VITA_GDB=1)
	target_compile_options(cdogs-sdl PRIVATE
		-O0 -g -fno-omit-frame-pointer -fno-inline-functions)
	target_compile_definitions(cdogs PRIVATE CDOGS_VITA_GDB=1)
	target_compile_options(cdogs PRIVATE
		-O0 -g -fno-omit-frame-pointer -fno-inline-functions)
endif()

if(CDOGS_PARTICLE_ATLAS)
	target_compile_definitions(cdogs-sdl PRIVATE CDOGS_PARTICLE_ATLAS=1)
	target_compile_definitions(cdogs PRIVATE CDOGS_PARTICLE_ATLAS=1)
endif()
if(CDOGS_TERRAIN_CACHE)
	target_compile_definitions(cdogs-sdl PRIVATE CDOGS_TERRAIN_CACHE=1)
	target_compile_definitions(cdogs PRIVATE CDOGS_TERRAIN_CACHE=1)
endif()

if(VITA)
	if(NOT DEFINED VITA_APP_NAME)
		set(VITA_APP_NAME "C-Dogs SDL")
	endif()
	if(NOT DEFINED VITA_TITLEID)
		set(VITA_TITLEID "CDOG00001")
	endif()
	if(NOT DEFINED VITA_VERSION)
		if(VERSION_MAJOR LESS 10)
			set(_vita_ver_maj "0${VERSION_MAJOR}")
		else()
			set(_vita_ver_maj "${VERSION_MAJOR}")
		endif()
		if(VERSION_MINOR LESS 10)
			set(_vita_ver_min "0${VERSION_MINOR}")
		else()
			set(_vita_ver_min "${VERSION_MINOR}")
		endif()
		set(VITA_VERSION "${_vita_ver_maj}.${_vita_ver_min}")
	endif()
	vita_create_self(cdogs-sdl.self cdogs-sdl)
	# LiveArea / bubble assets live under vita_pkg/ (generated by
	# scripts/generate_vita_livearea.py). Stock game data stays on
	# ux0:data/cdogs-sdl/ and is intentionally NOT packaged into the VPK.
	set(VITA_PKG_DIR "${CMAKE_SOURCE_DIR}/vita_pkg")
	vita_create_vpk(cdogs-sdl.vpk ${VITA_TITLEID} cdogs-sdl.self
		VERSION ${VITA_VERSION}
		NAME ${VITA_APP_NAME}
		FILE "${VITA_PKG_DIR}/sce_sys/icon0.png" sce_sys/icon0.png
		FILE "${VITA_PKG_DIR}/sce_sys/pic0.png" sce_sys/pic0.png
		FILE "${VITA_PKG_DIR}/sce_sys/livearea/contents/bg0.png" sce_sys/livearea/contents/bg0.png
		FILE "${VITA_PKG_DIR}/sce_sys/livearea/contents/startup.png" sce_sys/livearea/contents/startup.png
		FILE "${VITA_PKG_DIR}/sce_sys/livearea/contents/template.xml" sce_sys/livearea/contents/template.xml
	)
endif()
